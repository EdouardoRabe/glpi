# 3.1 React Router - Bases

## QUOI ?

**React Router** = navigation côté client sans rechargement de page. Les URLs changent, l'interface se met à jour.

```jsx
<BrowserRouter>
  <Routes>
    <Route path="/" element={<Home />} />
    <Route path="/about" element={<About />} />
  </Routes>
</BrowserRouter>
```

---

## Installation

```bash
npm install react-router-dom
```

---

## Modèle réutilisable

```jsx
import { BrowserRouter, Routes, Route, Link } from 'react-router-dom'

function App() {
  return (
    <BrowserRouter>
      <nav>
        <Link to="/">Accueil</Link>
        <Link to="/about">À propos</Link>
      </nav>

      <Routes>
        <Route path="/" element={<Home />} />
        <Route path="/about" element={<About />} />
      </Routes>
    </BrowserRouter>
  )
}

function Home() {
  return <h1>Accueil</h1>
}

function About() {
  return <h1>À propos</h1>
}

export default App
```

## Navigation

### Liens déclaratifs (préféré)

```jsx
import { Link, NavLink } from 'react-router-dom'

// Link simple
<Link to="/">Accueil</Link>

// NavLink avec classe active
<NavLink 
  to="/about"
  className={({ isActive }) => isActive ? 'active' : ''}
>
  À propos
</NavLink>
```

### Navigation impérative (programmée)

```jsx
import { useNavigate } from 'react-router-dom'

function LoginSuccess() {
  const navigate = useNavigate()

  const goHome = () => {
    navigate('/')
  }

  return <button onClick={goHome}>Retour accueil</button>
}
```

## EXEMPLES

### Exemple 1 : App multi-pages simple

```jsx
import { BrowserRouter, Routes, Route, Link } from 'react-router-dom'

function App() {
  return (
    <BrowserRouter>
      <nav>
        <Link to="/">Home</Link>
        <Link to="/users">Users</Link>
        <Link to="/products">Products</Link>
      </nav>

      <Routes>
        <Route path="/" element={<Home />} />
        <Route path="/users" element={<Users />} />
        <Route path="/products" element={<Products />} />
      </Routes>
    </BrowserRouter>
  )
}

function Home() { return <h1>Accueil</h1> }
function Users() { return <h1>Utilisateurs</h1> }
function Products() { return <h1>Produits</h1> }
```

### Exemple 2 : Route 404

```jsx
<Routes>
  <Route path="/" element={<Home />} />
  <Route path="/about" element={<About />} />
  <Route path="*" element={<NotFound />} />  {/* Toutes les autres URLs */}
</Routes>

function NotFound() {
  return <h1>❌ Page non trouvée</h1>
}
```
