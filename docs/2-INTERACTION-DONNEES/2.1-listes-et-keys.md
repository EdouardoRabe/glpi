# 2.1 Listes & Keys

## QUOI ?

Afficher une liste en React se fait avec `.map()`. Chaque élément doit avoir une `key` unique pour que React puisse les mettre à jour efficacement.

```jsx
const fruits = ['Pomme', 'Banane', 'Cerise'];

// ✅ BON : avec key
{fruits.map((fruit, index) => (
  <li key={index}>{fruit}</li>
))}
```

---

## POURQUOI ?

- ✅ Afficher des collections de données
- ✅ React optimise les mises à jour
- ✅ Éviter les bugs d'interface
- ✅ Conserver l'état des composants

---

## COMMENT ?

### Modèle réutilisable

```jsx
// ✅ IMPORTS NECESSAIRES
// (Rien à importer pour .map() !)

const items = ['item1', 'item2', 'item3'];

function ListeItems() {
  return (
    <ul>
      {items.map((item) => (
        <li key={item}>{item}</li>  {/* ← key unique */}
      ))}
    </ul>
  )
}

export default ListeItems
```

### Règle 1 : Utilise un ID unique, pas l'index

```jsx
// ❌ RISQUÉ : l'index peut changer
{items.map((item, index) => (
  <li key={index}>{item}</li>  // ❌ Pas idéal
))}

// ✅ BON : ID stable
{items.map((item) => (
  <li key={item.id}>{item.nom}</li>  // ✅ Recommandé
))}
```

**Quand utiliser l'index ?**
- Liste **statique** (jamais modifiée)
- Jamais ni triée, ni filtrée
- Pas d'ajout/suppression

### Règle 2 : La key doit être unique dans la liste

```jsx
// ❌ ERREUR : deux items avec la même key
[
  { id: 1, nom: 'Alice' },
  { id: 1, nom: 'Bob' }   // ← Même ID !
]

// ✅ BON : IDs uniques
[
  { id: 1, nom: 'Alice' },
  { id: 2, nom: 'Bob' }
]
```

## EXEMPLES

### Exemple 1 : Liste simple

```jsx
function FruitList() {
  const fruits = ['Pomme', 'Banane', 'Cerise', 'Fraise'];

  return (
    <ul>
      {fruits.map((fruit) => (
        <li key={fruit}>{fruit}</li>
      ))}
    </ul>
  )
}

export default FruitList
```

### Exemple 2 : Liste d'objets

```jsx
function UserList() {
  const users = [
    { id: 1, nom: 'Alice', email: 'alice@ex.com' },
    { id: 2, nom: 'Bob', email: 'bob@ex.com' },
    { id: 3, nom: 'Charlie', email: 'charlie@ex.com' }
  ];

  return (
    <ul>
      {users.map((user) => (
        <li key={user.id}>
          <strong>{user.nom}</strong> - {user.email}
        </li>
      ))}
    </ul>
  )
}

export default UserList
```

### Exemple 3 : Liste dynamique avec suppression

```jsx
import { useState } from 'react'

function TodoList() {
  const [todos, setTodos] = useState([
    { id: 1, texte: 'Faire les courses' },
    { id: 2, texte: 'Coder un projet' },
    { id: 3, texte: 'Faire du sport' }
  ])

  const supprimerTodo = (id) => {
    setTodos(todos.filter(todo => todo.id !== id))
  }

  return (
    <ul>
      {todos.map((todo) => (
        <li key={todo.id}>
          {todo.texte}
          <button onClick={() => supprimerTodo(todo.id)}>Supprimer</button>
        </li>
      ))}
    </ul>
  )
}

export default TodoList
```

### Exemple 4 : Liste imbriquée

```jsx
function GroupedUsers() {
  const groups = [
    {
      group: 'Admins',
      users: [
        { id: 1, nom: 'Alice' },
        { id: 2, nom: 'Bob' }
      ]
    },
    {
      group: 'Users',
      users: [
        { id: 3, nom: 'Charlie' },
        { id: 4, nom: 'Diana' }
      ]
    }
  ];

  return (
    <div>
      {groups.map((group) => (
        <div key={group.group}>
          <h3>{group.group}</h3>
          <ul>
            {group.users.map((user) => (
              <li key={user.id}>{user.nom}</li>
            ))}
          </ul>
        </div>
      ))}
    </div>
  )
}

export default GroupedUsers
```

### Exemple 5 : Liste avec composant

```jsx
function UserCard({ user }) {
  return (
    <div className="card">
      <h3>{user.nom}</h3>
      <p>{user.email}</p>
      <p>ID: {user.id}</p>
    </div>
  )
}

function UserGrid() {
  const users = [
    { id: 1, nom: 'Alice', email: 'alice@ex.com' },
    { id: 2, nom: 'Bob', email: 'bob@ex.com' },
    { id: 3, nom: 'Charlie', email: 'charlie@ex.com' }
  ];

  return (
    <div className="grid">
      {users.map((user) => (
        <UserCard key={user.id} user={user} />
      ))}
    </div>
  )
}

export default UserGrid
```

## Erreurs courantes

### ❌ Oublier la key

```jsx
// ❌ ERREUR : pas de key
{items.map((item) => (
  <li>{item.nom}</li>  // ❌ React affiche un warning
))}

// ✅ BON
{items.map((item) => (
  <li key={item.id}>{item.nom}</li>
))}
```

### ❌ Key qui change à chaque rendu

```jsx
// ❌ ERREUR : new key à chaque rendu
{items.map((item) => (
  <li key={Math.random()}>{item.nom}</li>  // Génère UUID à chaque fois!
))}

// ✅ BON : ID stable
{items.map((item) => (
  <li key={item.id}>{item.nom}</li>
))}
```

### ❌ Utiliser l'index sur liste dynamique

```jsx
// ❌ RISQUÉ : si la liste est triée/filtrée
const [users, setUsers] = useState([...])

setUsers(users.sort(...))  // Changement d'ordre

{users.map((user, index) => (  // Index = nouveau!
  <li key={index}>{user.nom}</li>  // ❌ Bug potentiel
))}
```
