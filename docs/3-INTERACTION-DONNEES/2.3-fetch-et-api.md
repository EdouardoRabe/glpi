# 2.3 Fetch & API

## QUOI ?

**Fetch** = fonction JavaScript pour faire des requêtes HTTP vers des serveurs/APIs.

**API** = une adresse web qui retourne des données (JSON).

```jsx
fetch('https://api.example.com/users')
  .then(res => res.json())
  .then(data => console.log(data))
```

---

## POURQUOI ?

- ✅ Récupérer des données d'un serveur
- ✅ Envoyer des données au serveur
- ✅ Charger du contenu dynamique
- ✅ Interagir avec des services externes

---

## COMMENT ?

### Modèle réutilisable

```jsx
import { useState, useEffect } from 'react'

function MonAPI() {
  const [data, setData] = useState(null)
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState(null)

  useEffect(() => {
    fetch('https://api.example.com/donnees')
      .then(res => res.json())
      .then(data => {
        setData(data)
        setLoading(false)
      })
      .catch(err => {
        setError(err)
        setLoading(false)
      })
  }, [])

  if (loading) return <p>Chargement...</p>
  if (error) return <p>Erreur: {error.message}</p>

  return <pre>{JSON.stringify(data, null, 2)}</pre>
}

export default MonAPI
```

### Important : Toujours utiliser useEffect!

```jsx
// ❌ ERREUR : fetch appelé à chaque rendu = boucle infinie!
function App() {
  fetch('https://api.example.com/data')
  return <div>...</div>
}

// ✅ BON : fetch dans useEffect
import { useEffect } from 'react'

function App() {
  useEffect(() => {
    fetch('https://api.example.com/data')
  }, [])
  return <div>...</div>
}
```

### Les trois états

Toujours gérer ces 3 états :

```jsx
const [data, setData] = useState(null)         // Les données
const [loading, setLoading] = useState(true)   // En cours?
const [error, setError] = useState(null)       // Une erreur?
```

| État | Affichage |
|------|-----------|
| `loading === true` | Spinner, "Chargement..." |
| `error !== null` | Message d'erreur |
| `data !== null` | Le contenu réel |

## EXEMPLES

### Exemple 1 : GET simple

```jsx
import { useState, useEffect } from 'react'

function ListeUtilisateurs() {
  const [users, setUsers] = useState([])
  const [loading, setLoading] = useState(true)

  useEffect(() => {
    fetch('https://jsonplaceholder.typicode.com/users')
      .then(res => res.json())
      .then(data => {
        setUsers(data)
        setLoading(false)
      })
  }, [])

  if (loading) return <p>Chargement...</p>

  return (
    <ul>
      {users.map(user => (
        <li key={user.id}>{user.name} - {user.email}</li>
      ))}
    </ul>
  )
}

export default ListeUtilisateurs
```

### Exemple 2 : GET avec dépendance

```jsx
import { useState, useEffect } from 'react'

function ProfilUtilisateur({ userId }) {
  const [user, setUser] = useState(null)
  const [loading, setLoading] = useState(true)

  useEffect(() => {
    fetch(`https://jsonplaceholder.typicode.com/users/${userId}`)
      .then(res => res.json())
      .then(data => {
        setUser(data)
        setLoading(false)
      })
  }, [userId])  // ← Se relance si userId change

  if (loading) return <p>Chargement...</p>

  return (
    <div>
      <h2>{user.name}</h2>
      <p>Email: {user.email}</p>
      <p>Téléphone: {user.phone}</p>
    </div>
  )
}

export default ProfilUtilisateur
```

### Exemple 3 : POST (envoyer des données)

```jsx
import { useState } from 'react'

function CreerUtilisateur() {
  const [nom, setNom] = useState('')
  const [loading, setLoading] = useState(false)
  const [error, setError] = useState(null)

  const handleSubmit = async (e) => {
    e.preventDefault()
    setLoading(true)

    try {
      const res = await fetch('https://jsonplaceholder.typicode.com/users', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ name: nom })
      })

      if (!res.ok) throw new Error('Erreur serveur')

      const data = await res.json()
      console.log('Utilisateur créé:', data)
      setNom('')
    } catch (err) {
      setError(err.message)
    } finally {
      setLoading(false)
    }
  }

  return (
    <form onSubmit={handleSubmit}>
      <input 
        value={nom}
        onChange={(e) => setNom(e.target.value)}
        placeholder="Nom"
      />
      <button disabled={loading}>{loading ? 'Envoi...' : 'Créer'}</button>
      {error && <p style={{ color: 'red' }}>{error}</p>}
    </form>
  )
}

export default CreerUtilisateur
```

### Exemple 4 : Avec gestion des erreurs complète

```jsx
import { useState, useEffect } from 'react'

function API() {
  const [data, setData] = useState(null)
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState(null)

  useEffect(() => {
    fetch('https://api.example.com/data')
      .then(res => {
        if (!res.ok) throw new Error(`Erreur ${res.status}`)
        return res.json()
      })
      .then(data => {
        setData(data)
        setLoading(false)
      })
      .catch(err => {
        setError(err.message)
        setLoading(false)
      })
  }, [])

  if (loading) return <p>⏳ Chargement...</p>
  if (error) return <p>❌ Erreur: {error}</p>
  if (!data) return <p>Aucune donnée</p>

  return (
    <div>
      <h2>Données reçues:</h2>
      <pre>{JSON.stringify(data, null, 2)}</pre>
    </div>
  )
}

export default API
```

## Syntaxe async/await (plus lisible)

```jsx
import { useState, useEffect } from 'react'

function APIAsyncAwait() {
  const [data, setData] = useState(null)
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState(null)

  useEffect(() => {
    const chargerData = async () => {
      try {
        const res = await fetch('https://api.example.com/data')
        if (!res.ok) throw new Error('Erreur serveur')
        
        const data = await res.json()
        setData(data)
      } catch (err) {
        setError(err.message)
      } finally {
        setLoading(false)
      }
    }

    chargerData()
  }, [])

  if (loading) return <p>Chargement...</p>
  if (error) return <p>Erreur: {error}</p>

  return <pre>{JSON.stringify(data, null, 2)}</pre>
}

export default APIAsyncAwait
```

## Les méthodes HTTP

| Méthode | Usage |
|---------|-------|
| `GET` | Récupérer des données |
| `POST` | Créer une nouvelle ressource |
| `PUT` | Remplacer une ressource entière |
| `PATCH` | Modifier partiellement une ressource |
| `DELETE` | Supprimer une ressource |

```jsx
// GET (par défaut)
fetch('https://api.example.com/users')

// POST
fetch('https://api.example.com/users', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({ nom: 'Alice' })
})

// DELETE
fetch('https://api.example.com/users/1', {
  method: 'DELETE'
})
```

## Pièges courants

### ❌ Oublier `.json()`

```jsx
// ❌ ERREUR : data n'est pas convertie
fetch('https://api.example.com/data')
  .then(res => console.log(res))  // Affiche Response, pas les données

// ✅ BON
fetch('https://api.example.com/data')
  .then(res => res.json())  // Convertit en JSON
  .then(data => console.log(data))
```

### ❌ Ne pas initialiser `loading` à `true`

```jsx
// ❌ Au premier rendu, loading=false alors qu'on charge !
const [loading, setLoading] = useState(false)

// ✅ BON
const [loading, setLoading] = useState(true)
```

### ❌ Mettre le fetch hors de useEffect

```jsx
// ❌ Boucle infinie !
function App() {
  const [data, setData] = useState(null)
  
  fetch('https://api.example.com/data')  // À chaque rendu
    .then(res => res.json())
    .then(data => setData(data))  // Déclenche un nouveau rendu

  return <div>...</div>
}

// ✅ BON : dans useEffect
useEffect(() => {
  fetch(...)
}, [])
```
