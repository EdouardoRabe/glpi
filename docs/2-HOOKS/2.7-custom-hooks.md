# 2.7 - Custom Hooks (Créer Ses Propres Hooks)

## QUOI ?
Un Custom Hook est une **fonction JavaScript** qui utilise d'autres hooks. C'est une façon de réutiliser de la logique entre composants sans les rendre imbriqués.

## POURQUOI ?
- Réutiliser la logique entre plusieurs composants
- Garder les composants simples et déclaratifs
- Suivre une convention claire: `use*`
- Abstraire la complexité

## COMMENT ?

### Hook Personnalisé Simple: useCounter

```jsx
import { useState, useCallback } from 'react';

// Custom Hook: C'est juste une fonction qui utilise des hooks
function useCounter(initialValue = 0) {
  const [count, setCount] = useState(initialValue);
  
  const increment = useCallback(() => {
    setCount(c => c + 1);
  }, []);
  
  const decrement = useCallback(() => {
    setCount(c => c - 1);
  }, []);
  
  const reset = useCallback(() => {
    setCount(initialValue);
  }, [initialValue]);
  
  return { count, increment, decrement, reset };
}

// Utilisation
function Counter() {
  const { count, increment, decrement, reset } = useCounter(0);
  
  return (
    <div>
      <p>{count}</p>
      <button onClick={increment}>+</button>
      <button onClick={decrement}>-</button>
      <button onClick={reset}>Reset</button>
    </div>
  );
}
```

### Hook Personnalisé: useFetch

```jsx
import { useState, useEffect } from 'react';

function useFetch(url) {
  const [data, setData] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);
  
  useEffect(() => {
    const fetchData = async () => {
      try {
        const response = await fetch(url);
        if (!response.ok) throw new Error('Fetch failed');
        const json = await response.json();
        setData(json);
      } catch (err) {
        setError(err.message);
      } finally {
        setLoading(false);
      }
    };
    
    fetchData();
  }, [url]);
  
  return { data, loading, error };
}

// Utilisation
function UsersList() {
  const { data: users, loading, error } = useFetch('/api/users');
  
  if (loading) return <p>Chargement...</p>;
  if (error) return <p>Erreur: {error}</p>;
  
  return (
    <ul>
      {users.map(user => (
        <li key={user.id}>{user.name}</li>
      ))}
    </ul>
  );
}
```

### Hook Personnalisé: useForm

```jsx
import { useState, useCallback } from 'react';

function useForm(initialValues, onSubmit) {
  const [values, setValues] = useState(initialValues);
  const [errors, setErrors] = useState({});
  
  const handleChange = useCallback((e) => {
    const { name, value } = e.target;
    setValues(prev => ({ ...prev, [name]: value }));
  }, []);
  
  const handleSubmit = useCallback((e) => {
    e.preventDefault();
    // Validation simple
    const newErrors = {};
    Object.entries(values).forEach(([key, val]) => {
      if (!val) newErrors[key] = `${key} est requis`;
    });
    
    if (Object.keys(newErrors).length === 0) {
      onSubmit(values);
    } else {
      setErrors(newErrors);
    }
  }, [values, onSubmit]);
  
  const reset = useCallback(() => {
    setValues(initialValues);
    setErrors({});
  }, [initialValues]);
  
  return {
    values,
    errors,
    handleChange,
    handleSubmit,
    reset,
    setValues
  };
}

// Utilisation
function LoginForm() {
  const { values, errors, handleChange, handleSubmit, reset } = useForm(
    { email: '', password: '' },
    (values) => {
      console.log('Form submitted:', values);
      reset();
    }
  );
  
  return (
    <form onSubmit={handleSubmit}>
      <input 
        name="email" 
        value={values.email}
        onChange={handleChange}
        placeholder="Email"
      />
      {errors.email && <p style={{ color: 'red' }}>{errors.email}</p>}
      
      <input 
        name="password"
        type="password"
        value={values.password}
        onChange={handleChange}
        placeholder="Password"
      />
      {errors.password && <p style={{ color: 'red' }}>{errors.password}</p>}
      
      <button type="submit">Se Connecter</button>
    </form>
  );
}
```

### Hook Personnalisé: useLocalStorage

```jsx
import { useState, useEffect } from 'react';

function useLocalStorage(key, initialValue) {
  // Lire depuis localStorage au démarrage
  const [value, setValue] = useState(() => {
    try {
      const item = window.localStorage.getItem(key);
      return item ? JSON.parse(item) : initialValue;
    } catch (error) {
      console.error(error);
      return initialValue;
    }
  });
  
  // Sauvegarder quand la valeur change
  useEffect(() => {
    try {
      window.localStorage.setItem(key, JSON.stringify(value));
    } catch (error) {
      console.error(error);
    }
  }, [key, value]);
  
  return [value, setValue];
}

// Utilisation
function Theme() {
  const [theme, setTheme] = useLocalStorage('theme', 'light');
  
  return (
    <div>
      <p>Thème: {theme}</p>
      <button onClick={() => setTheme(theme === 'light' ? 'dark' : 'light')}>
        Changer de thème (persiste!)
      </button>
    </div>
  );
}
```

### Hook Personnalisé: useDebounce

```jsx
import { useState, useEffect } from 'react';

function useDebounce(value, delay = 500) {
  const [debouncedValue, setDebouncedValue] = useState(value);
  
  useEffect(() => {
    const handler = setTimeout(() => {
      setDebouncedValue(value);
    }, delay);
    
    return () => clearTimeout(handler);
  }, [value, delay]);
  
  return debouncedValue;
}

// Utilisation
function Search() {
  const [searchTerm, setSearchTerm] = useState('');
  const debouncedSearchTerm = useDebounce(searchTerm, 300);
  
  useEffect(() => {
    if (debouncedSearchTerm) {
      console.log('Rechercher:', debouncedSearchTerm);
      // Faire l'appel API
    }
  }, [debouncedSearchTerm]);
  
  return (
    <input 
      value={searchTerm}
      onChange={(e) => setSearchTerm(e.target.value)}
      placeholder="Chercher..."
    />
  );
}
```

## Exemple Complet: useAsync

```jsx
import { useState, useEffect, useCallback } from 'react';

function useAsync(asyncFunction, immediate = true) {
  const [status, setStatus] = useState('idle');
  const [error, setError] = useState(null);
  const [data, setData] = useState(null);
  
  const execute = useCallback(async () => {
    setStatus('pending');
    setData(null);
    setError(null);
    
    try {
      const result = await asyncFunction();
      setData(result);
      setStatus('success');
      return result;
    } catch (err) {
      setError(err);
      setStatus('error');
    }
  }, [asyncFunction]);
  
  useEffect(() => {
    if (immediate) {
      execute();
    }
  }, [execute, immediate]);
  
  return { execute, status, data, error };
}

// Utilisation
function UserProfile({ userId }) {
  const { data: user, status, error, execute } = useAsync(
    () => fetch(`/api/user/${userId}`).then(r => r.json()),
    true  // Chercher immédiatement
  );
  
  if (status === 'pending') return <p>Chargement...</p>;
  if (status === 'error') return <p>Erreur: {error.message}</p>;
  if (status === 'success') return <p>Utilisateur: {user.name}</p>;
}
```

## Règles des Hooks Personnalisés

✅ **Règles à Respecter:**
- Commencer le nom par `use` (toujours!)
- Utiliser les hooks de React à l'intérieur
- Appeler les hooks au niveau top-level (pas dans les boucles/conditions)
- Retourner ce que tu veux (valeur, tableau, objet)

❌ **À Éviter:**
- Appeler les hooks conditionnellement
- Appeler les hooks dans des boucles
- Appeler les hooks dans des fonctions imbriquées
- Nommer trop génériquement (éviter `useData`)

## Pièges Courants

❌ **Oublier les dépendances**
```jsx
// ❌ userEffect re-crée la fonction à chaque rendu
function useUser(userId) {
  const fetchUser = async () => {  // Créée à chaque appel du hook!
    return fetch(`/api/user/${userId}`).then(r => r.json());
  };
  
  useEffect(() => {
    fetchUser();
  }, [fetchUser]);  // S'exécute à l'infini!
}

// ✅ Correct
function useUser(userId) {
  useEffect(() => {
    const fetchUser = async () => {
      return fetch(`/api/user/${userId}`).then(r => r.json());
    };
    fetchUser();
  }, [userId]);
}
```

❌ **Ne pas respecter les règles des hooks (appels conditionnels)**
```jsx
// ❌ ERREUR - appeler useState conditionnellement
function BAD(condition) {
  if (condition) {
    const [value, setValue] = useState(0);  // JAMAIS!
  }
}

// ✅ Correct - toujours à la même place
function GOOD(condition) {
  const [value, setValue] = useState(0);  // Toujours au top-level
}
```

## À Retenir ✨

- ✅ Un custom hook est une fonction qui utilise des React Hooks
- ✅ **Toujours** commencer par `use` (useCounter, useFetch, etc.)
- ✅ Les deux règles des hooks s'appliquent aussi aux custom hooks
- ✅ Retourner ce qui est réutilisable
- ✅ Testables et composables
- ✅ Excellent pour éviter les composants "wrapper"
