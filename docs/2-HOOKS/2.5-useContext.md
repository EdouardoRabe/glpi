# 2.5 - useContext (Partage Global de Données)

## QUOI ?
`useContext` permet de partager des données entre composants **sans passer par les props** (évite le "prop drilling"). Les données sont accessibles partout si on est à l'intérieur d'un Provider.

## POURQUOI ?
- Éviter de passer les props par 10 niveaux de composants
- Partager l'authentification (user global)
- Partager le thème (light/dark global)
- Partager les préférences utilisateur

## COMMENT ?

### Créer un Context

```jsx
import { createContext, useState } from 'react';

// 1. Créer le context
const ThemeContext = createContext();

// 2. Créer un Provider
export function ThemeProvider({ children }) {
  const [theme, setTheme] = useState('light');
  
  const toggleTheme = () => {
    setTheme(theme === 'light' ? 'dark' : 'light');
  };
  
  return (
    <ThemeContext.Provider value={{ theme, toggleTheme }}>
      {children}
    </ThemeContext.Provider>
  );
}

// 3. Exporter le context pour l'utiliser
export default ThemeContext;
```

### Utiliser un Context

```jsx
import { useContext } from 'react';
import ThemeContext from './ThemeContext';

function Header() {
  const { theme, toggleTheme } = useContext(ThemeContext);
  
  return (
    <header style={{ background: theme === 'light' ? 'white' : 'black' }}>
      <h1 style={{ color: theme === 'light' ? 'black' : 'white' }}>
        Mon Site
      </h1>
      <button onClick={toggleTheme}>
        Passer au {theme === 'light' ? 'dark' : 'light'} mode
      </button>
    </header>
  );
}

function Footer() {
  const { theme } = useContext(ThemeContext);
  
  return (
    <footer style={{ background: theme === 'light' ? 'white' : 'black' }}>
      <p>© 2024</p>
    </footer>
  );
}
```

### Utilisation dans App.jsx

```jsx
import { ThemeProvider } from './ThemeContext';
import Header from './Header';
import Footer from './Footer';

export default function App() {
  return (
    <ThemeProvider>
      <Header />
      <main>Contenu principal</main>
      <Footer />
    </ThemeProvider>
  );
}
```

## Exemple Complet: Context d'Authentification

```jsx
// 1. Créer le context
import { createContext, useState } from 'react';

const AuthContext = createContext();

export function AuthProvider({ children }) {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(false);
  
  const login = async (email, password) => {
    setLoading(true);
    try {
      const response = await fetch('/api/login', {
        method: 'POST',
        body: JSON.stringify({ email, password })
      });
      const userData = await response.json();
      setUser(userData);
    } catch (error) {
      console.error('Login failed');
    } finally {
      setLoading(false);
    }
  };
  
  const logout = () => {
    setUser(null);
  };
  
  return (
    <AuthContext.Provider value={{ user, login, logout, loading }}>
      {children}
    </AuthContext.Provider>
  );
}

export default AuthContext;

// 2. Utiliser dans les composants
import { useContext } from 'react';
import AuthContext from './AuthContext';

function LoginButton() {
  const { user, login, logout } = useContext(AuthContext);
  
  if (user) {
    return <button onClick={logout}>Déconnexion ({user.name})</button>;
  }
  
  return <button onClick={() => login('user@example.com', 'password')}>
    Connexion
  </button>;
}

// 3. Dans App
import { AuthProvider } from './AuthContext';

export default function App() {
  return (
    <AuthProvider>
      <LoginButton />
    </AuthProvider>
  );
}
```

## Exemple Complet: Notification Globale

```jsx
import { createContext, useState } from 'react';

const NotificationContext = createContext();

export function NotificationProvider({ children }) {
  const [notifications, setNotifications] = useState([]);
  
  const addNotification = (message, type = 'info') => {
    const id = Date.now();
    setNotifications(prev => [...prev, { id, message, type }]);
    
    // Auto-remove après 3 secondes
    setTimeout(() => {
      removeNotification(id);
    }, 3000);
  };
  
  const removeNotification = (id) => {
    setNotifications(prev => prev.filter(n => n.id !== id));
  };
  
  return (
    <NotificationContext.Provider value={{ addNotification, removeNotification }}>
      {children}
      
      {/* Notifications UI */}
      <div style={{ position: 'fixed', top: 10, right: 10, zIndex: 1000 }}>
        {notifications.map(notif => (
          <div 
            key={notif.id}
            style={{
              background: notif.type === 'error' ? 'red' : 'blue',
              color: 'white',
              padding: '10px',
              margin: '5px',
              borderRadius: '5px'
            }}
          >
            {notif.message}
          </div>
        ))}
      </div>
    </NotificationContext.Provider>
  );
}

export default NotificationContext;

// Utilisation
import { useContext } from 'react';
import NotificationContext from './NotificationContext';

function SaveButton() {
  const { addNotification } = useContext(NotificationContext);
  
  const handleSave = () => {
    try {
      // Sauvegarder...
      addNotification('Données sauvegardées!', 'success');
    } catch (error) {
      addNotification('Erreur lors de la sauvegarde', 'error');
    }
  };
  
  return <button onClick={handleSave}>Sauvegarder</button>;
}
```

## Bonnes Pratiques

### Créer un Hook Personnalisé pour le Context

```jsx
// ❌ À faire à chaque fois
const { user } = useContext(AuthContext);

// ✅ Créer un custom hook
export function useAuth() {
  return useContext(AuthContext);
}

// Maintenant on peut faire:
function Component() {
  const { user } = useAuth();
  return <p>{user.name}</p>;
}
```

### Combiner useReducer et useContext

```jsx
import { createContext, useReducer } from 'react';

const AppContext = createContext();

function appReducer(state, action) {
  switch (action.type) {
    case 'SET_USER':
      return { ...state, user: action.payload };
    case 'SET_THEME':
      return { ...state, theme: action.payload };
    default:
      return state;
  }
}

export function AppProvider({ children }) {
  const [state, dispatch] = useReducer(appReducer, {
    user: null,
    theme: 'light'
  });
  
  return (
    <AppContext.Provider value={{ state, dispatch }}>
      {children}
    </AppContext.Provider>
  );
}

// Utilisation
export function useApp() {
  return useContext(AppContext);
}
```

## Pièges Courants

❌ **Utiliser Context pour les données qui changent souvent**
```jsx
// ❌ Mauvais si la valeur change à chaque rendu
const value = { user, setUser };  // Nouvel objet à chaque rendu!

// ✅ Correct - mémoriser la valeur
const value = useMemo(() => ({ user, setUser }), [user]);
```

❌ **Ne pas utiliser un Provider**
```jsx
// ❌ ERREUR - useContext retourne undefined
function App() {
  return <MyComponent />;  // Pas de Provider!
}

// ✅ Correct
function App() {
  return (
    <MyProvider>
      <MyComponent />
    </MyProvider>
  );
}
```

❌ **Trop de contexts différents**
```jsx
// ❌ Trop de providers imbriqués
<AuthProvider>
  <ThemeProvider>
    <NotificationProvider>
      <LanguageProvider>
        <App />
      </LanguageProvider>
    </NotificationProvider>
  </ThemeProvider>
</AuthProvider>

// ✅ Combiner dans un seul Provider
<GlobalProvider>
  <App />
</GlobalProvider>
```

## À Retenir ✨

- ✅ `useContext` évite le prop drilling
- ✅ Créer un context: `createContext()`
- ✅ Utiliser un Provider: `<Context.Provider value={data}>`
- ✅ Récupérer: `useContext(TheContext)`
- ✅ Créer des custom hooks pour simplifier
- ✅ Utiliser `useMemo` pour optimiser les valeurs
