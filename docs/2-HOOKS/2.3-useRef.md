# 2.3 - useRef (Références Directes)

## QUOI ?
`useRef` crée une référence mutable qui persiste à travers les re-rendus. Contrairement à `useState`, modifier une ref ne cause pas de re-render.

## POURQUOI ?
- Accéder directement à un élément DOM (input, vidéo, etc.)
- Garder une valeur mutable sans causer de re-render
- Stocker des valeurs qui ne participent pas au rendu
- Mesurer des dimensions d'éléments

## COMMENT ?

### Accéder à un Input

```jsx
import { useRef } from 'react';

function TextInput() {
  const inputRef = useRef(null);
  
  const focusInput = () => {
    inputRef.current.focus();
  };
  
  return (
    <>
      <input ref={inputRef} type="text" />
      <button onClick={focusInput}>Focus sur l'input</button>
    </>
  );
}
```

### Stocker une Valeur qui ne Cause pas de Re-render

```jsx
import { useRef, useEffect } from 'react';

function Timer() {
  const countRef = useRef(0);
  
  useEffect(() => {
    const interval = setInterval(() => {
      countRef.current++;
      console.log('Compté:', countRef.current);  // Ne cause pas de re-render
    }, 1000);
    
    return () => clearInterval(interval);
  }, []);
  
  return <p>Regardez la console (le composant n'a pas re-rendu)</p>;
}
```

### Différence useRef vs useState

```jsx
function CounterComparison() {
  const [stateCount, setStateCount] = useState(0);
  const refCount = useRef(0);
  
  const handleStateClick = () => {
    setStateCount(stateCount + 1);  // Cause un re-render
  };
  
  const handleRefClick = () => {
    refCount.current++;  // Ne cause PAS de re-render
    console.log('Ref:', refCount.current);
  };
  
  return (
    <>
      <p>State: {stateCount} (visible change)</p>
      <p>Ref: {refCount.current} (ne change pas à l'écran)</p>
      
      <button onClick={handleStateClick}>State +1</button>
      <button onClick={handleRefClick}>Ref +1 (voir console)</button>
    </>
  );
}
```

### Accéder aux Propriétés d'un Élément

```jsx
function TextAreaWithCharCount() {
  const textareaRef = useRef(null);
  const [charCount, setCharCount] = useState(0);
  
  const handleChange = () => {
    if (textareaRef.current) {
      setCharCount(textareaRef.current.value.length);
    }
  };
  
  return (
    <>
      <textarea 
        ref={textareaRef} 
        onChange={handleChange}
        placeholder="Tapez quelque chose..."
      />
      <p>Caractères: {charCount}</p>
    </>
  );
}
```

### Stocker l'ID d'un Interval/Timeout

```jsx
import { useRef, useEffect, useState } from 'react';

function StopWatch() {
  const [time, setTime] = useState(0);
  const intervalRef = useRef(null);
  
  const start = () => {
    intervalRef.current = setInterval(() => {
      setTime(t => t + 1);
    }, 100);
  };
  
  const stop = () => {
    clearInterval(intervalRef.current);
  };
  
  const reset = () => {
    clearInterval(intervalRef.current);
    setTime(0);
  };
  
  return (
    <>
      <p>Temps: {time * 0.1}s</p>
      <button onClick={start}>Démarrer</button>
      <button onClick={stop}>Arrêter</button>
      <button onClick={reset}>Réinitialiser</button>
    </>
  );
}
```

## Exemple Complet: Lecteur Vidéo Personnalisé

```jsx
import { useRef, useState } from 'react';

function VideoPlayer() {
  const videoRef = useRef(null);
  const [isPlaying, setIsPlaying] = useState(false);
  
  const togglePlay = () => {
    if (videoRef.current) {
      if (isPlaying) {
        videoRef.current.pause();
      } else {
        videoRef.current.play();
      }
      setIsPlaying(!isPlaying);
    }
  };
  
  return (
    <div>
      <video 
        ref={videoRef} 
        width="300"
        src="https://example.com/video.mp4"
      />
      <button onClick={togglePlay}>
        {isPlaying ? 'Pause' : 'Play'}
      </button>
    </div>
  );
}
```

## Exemple Complet: Focus Automatique sur Erreur

```jsx
import { useRef, useState } from 'react';

function FormWithAutoFocus() {
  const emailRef = useRef(null);
  const [error, setError] = useState('');
  
  const handleSubmit = (e) => {
    e.preventDefault();
    const email = emailRef.current.value;
    
    if (!email.includes('@')) {
      setError('Email invalide');
      emailRef.current.focus();
      emailRef.current.style.borderColor = 'red';
      return;
    }
    
    alert('Succès!');
  };
  
  return (
    <form onSubmit={handleSubmit}>
      <input ref={emailRef} type="email" placeholder="Email" />
      {error && <p style={{ color: 'red' }}>{error}</p>}
      <button type="submit">Envoyer</button>
    </form>
  );
}
```

## Pièges Courants

❌ **Utiliser ref.current directement dans le rendu**
```jsx
// ❌ Ne pas faire ça en JSX
function Bad() {
  const ref = useRef(0);
  return <p>{ref.current}</p>;  // Ne change JAMAIS à l'écran
}

// ✅ Utiliser setState si on veut que ça s'affiche
function Good() {
  const [count, setCount] = useState(0);
  return <p>{count}</p>;  // Change à l'affichage
}
```

❌ **Accéder à .current sur un ref non initialisé**
```jsx
// ❌ ref.current est null!
const inputRef = useRef(null);
inputRef.current.focus();  // CRASH!

// ✅ Vérifier d'abord
if (inputRef.current) {
  inputRef.current.focus();
}
```

❌ **Créer une ref dans le rendu**
```jsx
// ❌ MAUVAIS - crée une nouvelle ref à chaque rendu
function Bad() {
  const ref = useRef(null);  // Dans le corps du composant c'est ok
  
  // ❌ Mais pas dans une fonction
  const init = () => {
    const ref = useRef(null);  // Nouvelle ref à chaque appel!
  };
}
```

## À Retenir ✨

- ✅ `useRef` crée une référence persistante
- ✅ Modifier `.current` ne cause PAS de re-render
- ✅ Utiliser pour accéder directement au DOM
- ✅ Utiliser pour stocker des valeurs mutables
- ✅ ⚠️ Ne pas afficher `.current` directement (ça change pas à l'écran)
- ✅ Toujours vérifier que `.current` n'est pas null
